import sys, time
import pytest

@pytest.mark.parametrize("failures,expected", [(0, 60), (1, 5), (3, 20), (4, 40)])
def test_backoff(failures, expected):
    print(f"stdout for {failures}")
    print(f"stderr for {failures}", file=sys.stderr)
    assert failures != 3, f"case {failures}/{expected} deliberately fails"

class TestGroup:
    @pytest.mark.parametrize("value", ["a(b)[c].d", "", "  ", None])
    def test_names(self, value):
        assert value != "  "

    def test_plain(self):
        print("plain pass output")

    @pytest.mark.skip(reason="deliberately skipped")
    def test_skipped(self):
        pass

    def test_slow(self):
        time.sleep(1.2)
